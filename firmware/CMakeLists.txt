cmake_minimum_required(VERSION 3.16)
set(PROJECT_VER "0.0.0-dev")
if(DEFINED ENV{ZC_FW_VERSION})
    set(PROJECT_VER "$ENV{ZC_FW_VERSION}")
endif()
get_filename_component(ZC_BUILD_PROFILE "${SDKCONFIG}" NAME)
string(REPLACE "sdkconfig." "" ZC_BUILD_PROFILE "${ZC_BUILD_PROFILE}")
if(NOT ZC_BUILD_PROFILE MATCHES "^(inlet|outlet|control|inlet_real|outlet_real)$")
    message(FATAL_ERROR "Select a PlatformIO inlet, outlet, control, inlet_real or outlet_real environment")
endif()
# A real-data build keeps its role's project name (zcharmc-inlet/outlet): OTA and the
# release tools check the embedded name against the role, not the data mode.
string(REGEX REPLACE "_real$" "" ZC_BUILD_ROLE "${ZC_BUILD_PROFILE}")
include($ENV{IDF_PATH}/tools/cmake/project.cmake)
project(zcharmc-${ZC_BUILD_ROLE})
